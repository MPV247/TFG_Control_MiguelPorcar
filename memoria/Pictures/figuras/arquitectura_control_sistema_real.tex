\documentclass[tikz]{standalone}
\usetikzlibrary{positioning, arrows.meta, calc, fit, backgrounds, shapes.geometric}

\begin{document}
	
	\begin{tikzpicture}[
		node distance=1cm and 1.5cm,
		% Estilos
		bloque/.style={draw, thick, rectangle, minimum height=1.2cm, minimum width=2cm, align=center, font=\sffamily},
		periferico/.style={draw, thick, rectangle, minimum height=1cm, minimum width=1.5cm, align=center, font=\scriptsize\sffamily, fill=gray!10},
		planta/.style={draw, thick, rectangle, minimum height=1.2cm, minimum width=2.5cm, align=center, font=\sffamily, fill=green!5},
		flecha/.style={-{Stealth[scale=1.2]}, thick},
		etiqueta/.style={font=\tiny\sffamily}
		]
		
		% --- PROCESADOR (Controlador Digital) ---
		\node[bloque] (control) {Controlador Digital};
		\draw[flecha] ([xshift=-1.5cm]control.west) -- node[etiqueta, above] {$r_k$} (control.west);
		
		% --- PERIFÉRICOS DE SALIDA (Digital to Analog / PWM) ---
		\node[periferico, right=of control] (pwm) {ePWM \\ (DAC)};
		\node[periferico, right=1.2cm of pwm] (retenedor) {Retenedor \\ (ZOH)};
		
		% --- PROCESO CONTINUO (Planta y Driver) ---
		\node[planta, right=1.2cm of retenedor] (proceso) {Proceso Continuo \\ \scriptsize (Driver + Motor + Péndulo)};
		
		% --- SENSOR Y FEEDBACK ---
		\node[bloque, below=1.5cm of proceso, minimum width=2.5cm] (sensor) {Sensores \\ \scriptsize (Encoders)};
		
		% --- PERIFÉRICOS DE ENTRADA (Sampling / eQEP) ---
		\node[periferico, left=of sensor] (muestreo) {Muestreo};
		\node[periferico, left=of muestreo] (eqep) {eQEP \\ (ADC)};
		
		% --- CONEXIONES ---
		
		% Camino de ida
		\draw[flecha] (control.east) -- node[etiqueta, above] {$u_k$} (pwm.west);
		\draw[flecha] (pwm.east) -- node[etiqueta, above] {$u^*(kT)$} (retenedor.west);
		\draw[flecha] (retenedor.east) -- node[etiqueta, above] {$u(t)$} (proceso.west);
		
		% Camino de vuelta
		\draw[flecha] (proceso.east) -- ++(0.5,0) |- node[etiqueta, pos=0.25, right] {$y_p(t)$} (sensor.east);
		\draw[flecha] (sensor.west) -- node[etiqueta, above] {$y(t)$} (muestreo.east);
		\draw[flecha] (muestreo.west) -- node[etiqueta, above] {$y^*(kT)$} (eqep.east);
		\draw[flecha] (eqep.west) -| node[etiqueta, pos=0.8, left] {$y_k$} (control.south);
		
		% --- CAJAS DE AGRUPACIÓN ---
		\begin{scope}[on background layer]
			% Caja Procesador
			\node[draw, dashed, gray, fit=(control), inner sep=10pt, label={[gray, font=\sffamily]above:Procesador}] (caja_proc) {};
			
			% Caja Periféricos
			\node[draw, dashed, gray, fit=(pwm) (retenedor) (muestreo) (eqep), inner sep=16pt, label={[gray, font=\sffamily]above:Periféricos}] (caja_perif) {};
		\end{scope}
		
		% Dibujo del interruptor en el muestreo
		\draw[thick] ([xshift=-0.2cm, yshift=0.2cm]muestreo.north) -- ([xshift=0.2cm, yshift=0.1cm]muestreo.north);
		\node[etiqueta] at ([yshift=0.35cm]muestreo.north) {$T$};
		
	\end{tikzpicture}
